CREATE DEFINER=`root`@`localhost` PROCEDURE `get_courses_by_instructor` (
  IN p_instructor_id INT
)
BEGIN
  DECLARE v_count INT;

  SELECT COUNT(*) INTO v_count
  FROM Courses
  WHERE instructor_id = p_instructor_id;

  IF v_count = 0 THEN
    SELECT CONCAT('No courses found for instructor ID ', p_instructor_id) AS message;
  ELSE
    SELECT 
      course_id,
      title,
      description,
      price,
      created_at,
      last_updated,
      category_id
    FROM Courses
    WHERE instructor_id = p_instructor_id;
  END IF;
END




CREATE DEFINER=`root`@`localhost` PROCEDURE `get_enrolled_courses_by_student` (
  IN p_student_id INT
)
BEGIN
  DECLARE v_count INT;

  SELECT COUNT(*) INTO v_count
  FROM Enrollments
  WHERE user_id = p_student_id;

  IF v_count = 0 THEN
    SELECT CONCAT('No enrolled courses found for student ID ', p_student_id) AS message;
  ELSE
    SELECT 
      c.course_id,
      c.title,
      c.description,
      c.price,
      c.created_at,
      c.last_updated,
      c.category_id,
      e.enrolled_on,
      e.progress
    FROM Enrollments e
    JOIN Courses c ON e.course_id = c.course_id
    WHERE e.user_id = p_student_id;
  END IF;
END


CREATE PROCEDURE enroll_in_course_with_payment(
    IN p_user_id INT,
    IN p_course_id INT,
    IN p_payment_amount DECIMAL(10,2)
)
BEGIN
    DECLARE v_price DECIMAL(10,2);
    DECLARE v_user_exists INT;
    DECLARE v_course_exists INT;
    DECLARE v_enrolled INT;

    main_block: BEGIN

        -- Check if user exists
        SELECT COUNT(*) INTO v_user_exists FROM Users WHERE user_id = p_user_id;
        IF v_user_exists = 0 THEN
            SELECT 'User does not exist' AS message;
            LEAVE main_block;
        END IF;

        -- Check if course exists
        SELECT COUNT(*) INTO v_course_exists FROM Courses WHERE course_id = p_course_id;
        IF v_course_exists = 0 THEN
            SELECT 'Course does not exist' AS message;
            LEAVE main_block;
        END IF;

        -- Get course price
        SELECT price INTO v_price FROM Courses WHERE course_id = p_course_id;

        -- Check if already enrolled
        SELECT COUNT(*) INTO v_enrolled FROM Enrollments WHERE user_id = p_user_id AND course_id = p_course_id;
        IF v_enrolled > 0 THEN
            SELECT 'User already enrolled in this course' AS message;
            LEAVE main_block;
        END IF;

        -- Check if payment is correct
        IF p_payment_amount != v_price THEN
            SELECT 'Payment amount is not equal to course price' AS message;
            LEAVE main_block;
        END IF;

        -- Insert into Enrollments
        INSERT INTO Enrollments (user_id, course_id) VALUES (p_user_id, p_course_id);

        -- Insert into Payments
        INSERT INTO Payments (user_id, course_id, amount_paid) VALUES (p_user_id, p_course_id, p_payment_amount);

        SELECT 'Enrolled in course successfully' AS message;

    END main_block;
END;

CREATE PROCEDURE write_review (
    IN p_user_id INT,
    IN p_course_id INT,
    IN p_rating TINYINT,
    IN p_comment VARCHAR(1000)
)
BEGIN
    DECLARE v_enrolled INT;
    DECLARE v_user_exists INT;
    DECLARE v_course_exists INT;

    -- Check if the user exists
    SELECT COUNT(*) INTO v_user_exists
    FROM Users
    WHERE user_id = p_user_id;

    -- Check if the course exists
    SELECT COUNT(*) INTO v_course_exists
    FROM Courses
    WHERE course_id = p_course_id;

    -- If user or course doesn't exist, raise an error
    IF v_user_exists = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'User does not exist. Review not submitted.';
    ELSEIF v_course_exists = 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Course does not exist. Review not submitted.';
    ELSE
        -- Check if the user is enrolled in the course
        SELECT COUNT(*) INTO v_enrolled
        FROM Enrollments
        WHERE user_id = p_user_id AND course_id = p_course_id;

        -- If user is enrolled, allow them to write the review
        IF v_enrolled > 0 THEN
            -- Insert the review into the Reviews table
            INSERT INTO Reviews (user_id, course_id, rating, comment, review_date)
            VALUES (p_user_id, p_course_id, p_rating, p_comment, CURRENT_TIMESTAMP);
        ELSE
            -- If user is not enrolled, raise an error
            SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'User is not enrolled in this course. Review not submitted.';
        END IF;
    END IF;

END$$

CREATE TRIGGER trg_review_rating_check
BEFORE INSERT ON Reviews
FOR EACH ROW
BEGIN
    IF NEW.rating < 1 OR NEW.rating > 5 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Rating must be between 1 and 5.';
    END IF;
END

CREATE TRIGGER trg_review_rating_check_update
BEFORE UPDATE ON Reviews
FOR EACH ROW
BEGIN
    IF NEW.rating < 1 OR NEW.rating > 5 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Rating must be between 1 and 5.';
    END IF;
END


CREATE FUNCTION get_course_rating(p_course_id INT)
RETURNS DECIMAL(3,2)
DETERMINISTIC
BEGIN
    DECLARE v_avg_rating DECIMAL(3,2);

    SELECT AVG(rating) INTO v_avg_rating
    FROM Reviews
    WHERE course_id = p_course_id;

    RETURN v_avg_rating;
END

DROP PROCEDURE get_module_content;
CREATE DEFINER=`root`@`localhost` PROCEDURE `get_module_content`(IN p_module_id INT)
BEGIN
  -- Declare variables first
  DECLARE v_module_exists INT;

  -- Declare handler after variables
  DECLARE EXIT HANDLER FOR SQLEXCEPTION
  BEGIN
    SELECT 'An unexpected error occurred while retrieving module content.' AS error_message;
    ROLLBACK;
  END;

  -- Start a labeled block
  main_block: BEGIN

    -- Check if the module exists
    SELECT COUNT(*) INTO v_module_exists FROM Modules WHERE module_id = p_module_id;

    IF v_module_exists = 0 THEN
      SELECT 'Module not found' AS message;
      LEAVE main_block;
    END IF;

    -- If module exists, return data

    -- 1. Quizzes
    SELECT 
      quiz_id, 
      question_text, 
      correct_answer 
    FROM Quizzes 
    WHERE module_id = p_module_id;

    -- 2. Videos
    SELECT 
      video_id, 
      title, 
      position, 
      video_url 
    FROM Video 
    WHERE module_id = p_module_id;

    -- 3. Lesson Attachments
    SELECT 
      attachment_id, 
      position, 
      file_url, 
      uploaded_at 
    FROM Lesson_Attachments 
    WHERE module_id = p_module_id;

  END main_block;

END
