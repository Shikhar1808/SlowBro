-- USERS Table
CREATE TABLE Users (
  user_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  userName       VARCHAR2(100),
  email           VARCHAR2(100) UNIQUE,
  password        VARCHAR2(100),
  role            VARCHAR2(20) CHECK (role IN ('student', 'instructor', 'admin')),
  created_at      DATE DEFAULT SYSDATE
);

-- CATEGORIES Table
CREATE TABLE Categories (
  category_id   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  category_name   VARCHAR2(100) UNIQUE,
  description     VARCHAR2(255)
);

-- COURSES Table
CREATE TABLE Courses (
  course_id     NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  title           VARCHAR2(100),
  description     CLOB,
  price           NUMBER CHECK (price >= 0),
  created_at      DATE DEFAULT SYSDATE,
  last_updated    DATE DEFAULT SYSDATE,
  category_id     NUMBER REFERENCES Categories(category_id),
  instructor_id   NUMBER REFERENCES Users(user_id)
);

-- ENROLLMENTS Table
CREATE TABLE Enrollments (
  --enrollment_id   NUMBER PRIMARY KEY,
  user_id         NUMBER REFERENCES Users(user_id),
  course_id       NUMBER REFERENCES Courses(course_id),
  enrolled_on     DATE DEFAULT SYSDATE,
  progress        NUMBER DEFAULT 0 CHECK (progress BETWEEN 0 AND 100),
  PRIMARY KEY (user_id, course_id)
);

-- PAYMENTS Table
CREATE TABLE Payments (
    payment_id    NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id         NUMBER REFERENCES Users(user_id),
  course_id       NUMBER REFERENCES Courses(course_id),
  payment_date    DATE DEFAULT SYSDATE,
  amount_paid     NUMBER,
  --payment_status  VARCHAR2(20) CHECK (payment_status IN ('Success', 'Pending', 'Failed'))
);

-- REVIEWS Table
CREATE TABLE Reviews (
  review_id   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id         NUMBER REFERENCES Users(user_id),
  course_id       NUMBER REFERENCES Courses(course_id),
  rating          NUMBER CHECK (rating BETWEEN 1 AND 5),
  comment         VARCHAR2(1000),
  review_date     DATE DEFAULT SYSDATE
);

-- CERTIFICATES Table
CREATE TABLE Certificates (
  certificate_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id         NUMBER REFERENCES Users(user_id),
  course_id       NUMBER REFERENCES Courses(course_id),
  issued_on       DATE DEFAULT SYSDATE,
  is_issued       CHAR(1) DEFAULT 'N' CHECK (is_issued IN ('Y', 'N'))
);

-- MODULES Table
CREATE TABLE Modules (
 module_id   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  course_id       NUMBER REFERENCES Courses(course_id),
  title           VARCHAR2(100),
  position        NUMBER
);

-- Video Table
CREATE TABLE Video (
  video_id       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  module_id       NUMBER REFERENCES Modules(module_id),
  title           VARCHAR2(100),
  video_duration  NUMBER
);

-- LESSON_ATTACHMENTS Table
CREATE TABLE Lesson_Attachments (
  attachment_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  module_id       NUMBER REFERENCES Modules(module_id),
  --section_id      NUMBER REFERENCES Sections(section_id),
  file_name       VARCHAR2(200),
  file_type       VARCHAR2(50),
  uploaded_at     DATE DEFAULT SYSDATE
);

-- QUIZZES Table
CREATE TABLE Quizzes (
  quiz_id        NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  module_id       NUMBER REFERENCES Modules(module_id),
  --section_id      NUMBER REFERENCES Sections(section_id),
  question_text   VARCHAR2(1000),
  correct_answer  VARCHAR2(100)
);

-- QUIZ_RESULTS Table
CREATE TABLE Quiz_Results (
  result_id     NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  quiz_id         NUMBER REFERENCES Quizzes(quiz_id),
  user_id         NUMBER REFERENCES Users(user_id),
  --selected_answer VARCHAR2(100),
  --is_correct      CHAR(1) CHECK (is_correct IN ('Y', 'N')),
  score NUMBER(5,2),
  attempt_date    DATE DEFAULT SYSDATE
);



-- SEQUENCES
CREATE SEQUENCE CERTIFICATES_SEQ START WITH 1 INCREMENT BY 1;

-- TRIGGER: Update COURSES.last_updated on modification
CREATE OR REPLACE TRIGGER trg_course_last_updated
BEFORE UPDATE ON Courses
FOR EACH ROW
BEGIN
  :NEW.last_updated := SYSDATE;
END;
/

-- PROCEDURE: Award certificate if progress is 100%
CREATE OR REPLACE PROCEDURE award_certificate (
  p_user_id   IN NUMBER,
  p_course_id IN NUMBER
) AS
  v_progress Enrollments.progress%TYPE;
  v_exists   NUMBER := 0;
BEGIN
  SELECT progress INTO v_progress
  FROM Enrollments
  WHERE user_id = p_user_id AND course_id = p_course_id;

  SELECT COUNT(*) INTO v_exists
  FROM Certificates
  WHERE user_id = p_user_id AND course_id = p_course_id AND is_issued = 'Y';

  IF v_progress = 100 THEN
    IF v_exists = 0 THEN
      INSERT INTO Certificates (
        certificate_id, user_id, course_id, issued_on, is_issued
      ) VALUES (
        CERTIFICATES_SEQ.NEXTVAL, p_user_id, p_course_id, SYSDATE, 'Y'
      );
      DBMS_OUTPUT.PUT_LINE('Certificate awarded to user ID ' || p_user_id);
    ELSE
      DBMS_OUTPUT.PUT_LINE('Certificate already issued.');
    END IF;
  ELSE
    DBMS_OUTPUT.PUT_LINE('Course not completed. Progress = ' || v_progress);
  END IF;
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    DBMS_OUTPUT.PUT_LINE('No enrollment found.');
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END;
/
